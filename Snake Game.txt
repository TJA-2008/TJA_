def load_progress():
    global best_score, apples, unlocked_skins, selected_skin
    if os.path.exists(BEST_SCORE_FILE):
        try:
            with open(BEST_SCORE_FILE, "r", encoding="utf-8") as file:
                data = json.load(file)
                best_score = int(data.get("best_score", 0))
                apples = max(0, int(data.get("apples", 0)))
                stored_skins = data.get("unlocked_skins", ["Forest"])
                unlocked_skins = {
                    skin for skin in stored_skins if skin in SKINS
                } | {"Forest"}
                stored_selection = data.get("selected_skin", "Forest")
                if stored_selection in unlocked_skins:
                    selected_skin = stored_selection
        except (ValueError, TypeError, OSError):
            best_score = 0
            apples = 0
            unlocked_skins = {"Forest"}
            selected_skin = "Forest"
    apples_label.config(text=f"Apples: {apples}")


def save_progress():
    try:
        with open(BEST_SCORE_FILE, "w", encoding="utf-8") as file:
            json.dump(
                {
                    "best_score": best_score,
                    "apples": apples,
                    "unlocked_skins": sorted(unlocked_skins),
                    "selected_skin": selected_skin,
                },
                file,
            )
    except OSError:
        messagebox.showerror(
            "Save error",
            "Your latest score, apples, or skin selection could not be saved.",
            parent=window,
        )


def play_sound(kind):
    if kind == "eat":
        winsound.Beep(660, 80)
    elif kind == "game_over":
        winsound.Beep(220, 200)
        winsound.Beep(180, 220)


def get_speed():
    return max(MIN_SPEED, BASE_SPEED - score * SPEED_INCREASE_PER_POINT)

def open_skin_shop():
    global skin_shop_window
    if state == "playing":
        return
    if skin_shop_window is not None and skin_shop_window.winfo_exists():
        skin_shop_window.lift()
        return

    skin_shop_window = shop
    shop.title("Snake Skin Shop")
    shop.configure(bg=BACKGROUND_COLOR)
    shop.resizable(False, False)
    shop.transient(window)
    shop.grab_set()

    def close_shop():
        global skin_shop_window
        if shop.winfo_exists():
            shop.grab_release()
            shop.destroy()
        skin_shop_window = None

    shop.protocol("WM_DELETE_WINDOW", close_shop)
    Label(
        shop,

def buy_or_equip_skin(name, refresh_shop):
    global apples, selected_skin
    cost = SKINS[name]["cost"]
    if name not in unlocked_skins:
        if apples < cost:
            messagebox.showinfo(
                "Not enough apples",
                f"You need {cost - apples} more apples to unlock {name}.",
                parent=skin_shop_window,
            )
            return
        apples -= cost
        unlocked_skins.add(name)

    selected_skin = name
    apples_label.config(text=f"Apples: {apples}")
    save_progress()
    refresh_shop()
    if "snake" in globals():
        snake.draw()


class Snake:
    def __init__(self):
        self.coordinates = []
        self.items = []

        start_x = GAME_WIDTH // 2
        start_y = GAME_HEIGHT // 2
        for i in range(BODY_PARTS):
            self.coordinates.append([start_x - i * SPACE_SIZE, start_y])
        self.draw()

    def draw(self):
        palette = SKINS[selected_skin]
        for item in self.items:
            canvas.delete(item)
        self.items = []

        centers = [
            (x + SPACE_SIZE // 2, y + SPACE_SIZE // 2)
            for x, y in self.coordinates
        ]
        path = [coordinate for point in centers for coordinate in point]

        if len(centers) > 1:
            shadow_path = [
                coordinate
                for x, y in centers
                for coordinate in (x + 2, y + 5)
            ]
            self.items.append(
                canvas.create_line(
                    *shadow_path,
                    fill="#17261D",
                    width=32,
                    capstyle=ROUND,
                    joinstyle=ROUND,
                    smooth=True,
                    splinesteps=12,
                    tag="snake",
                )
            )
            self.items.append(
                canvas.create_line(
                    *path,
                    fill=palette["outline"],
                    width=34,
                    capstyle=ROUND,
                    joinstyle=ROUND,
                    smooth=True,
                    splinesteps=12,
                    tag="snake",
                )
            )
            self.items.append(
                canvas.create_line(
                    *path,
                    fill=palette["body"],
                    width=28,
                    capstyle=ROUND,
                    joinstyle=ROUND,
                    smooth=True,
                    splinesteps=12,
                    tag="snake",
                )
            )

        for index, (x, y) in enumerate(self.coordinates[1:-1], start=1):
            previous = centers[index - 1]
            following = centers[index + 1]
            dx = following[0] - previous[0]
            dy = following[1] - previous[1]
            if abs(dx) >= abs(dy):
                stripe_points = (x + 25, y + 12, x + 25, y + 38)
                belly_points = (x + 18, y + 37, x + 32, y + 37)
            else:
                stripe_points = (x + 12, y + 25, x + 38, y + 25)
                belly_points = (x + 37, y + 18, x + 37, y + 32)

            self.items.append(
                canvas.create_line(
                    *stripe_points,
                    fill=palette["scale"],
                    width=2,
                    smooth=True,
                    tag="snake",
                )
            )
            self.items.append(
                canvas.create_line(
                    *belly_points,
                    fill=palette["belly"],
                    width=5,
                    capstyle=ROUND,
                    tag="snake",
                )
            )

        if len(centers) > 1:
            first_x, first_y = centers[0]
            second_x, second_y = centers[1]
            tail_x, tail_y = centers[-1]
            before_tail_x, before_tail_y = centers[-2]
            head_x, head_y = self.coordinates[0]
            if abs(first_x - second_x) >= abs(first_y - second_y):
                tail_side = -1 if first_x > second_x else 1
                tail_points = (
                    tail_x,
                    tail_y - 7,
                    tail_x + (12 if tail_x > before_tail_x else -12),
                    tail_y,
                    tail_x,
                    tail_y + 7,
                )
                eye_positions = (
                    (first_x, first_y - 8),
                    (first_x, first_y + 8),
                )
                muzzle_box = (
                    head_x + (25 if tail_side < 0 else 5),
                    head_y + 13,
                    head_x + (47 if tail_side < 0 else 27),
                    head_y + 37,
                )
                nose_position = (
                    head_x + (8 if tail_side < 0 else 42),
                    head_y + 25,
                )
            else:
                tail_side = -1 if first_y > second_y else 1
                tail_points = (
                    tail_x - 7,
                    tail_y,
                    tail_x,
                    tail_y + (12 if tail_y > before_tail_y else -12),
                    tail_x + 7,
                    tail_y,
                )
                eye_positions = (
                    (first_x - 8, first_y),
                    (first_x + 8, first_y),
                )
                muzzle_box = (
                    head_x + 13,
                    head_y + (5 if tail_side < 0 else 25),
                    head_x + 37,
                    head_y + (27 if tail_side < 0 else 47),
                )
                nose_position = (
                    head_x + 25,
                    head_y + (8 if tail_side < 0 else 42),
                )

            self.items.append(
                canvas.create_polygon(
                    *tail_points,
                    fill=palette["body"],
                    outline=palette["outline"],
                    width=1,
                    smooth=True,
                    tag="snake",
                )
            )
            self.items.append(
                canvas.create_oval(
                    head_x + 5,
                    head_y + 5,
                    head_x + SPACE_SIZE - 5,
                    head_y + SPACE_SIZE - 5,
                    fill=palette["head"],
                    outline=palette["outline"],
                    width=2,
                    tag="snake",
                )
            )
            self.items.append(
                canvas.create_oval(
                    *muzzle_box,
                    fill=palette["body"],
                    outline=palette["outline"],
                    width=1,
                    tag="snake",
                )
            )
            for eye_x, eye_y in eye_positions:
                self.items.append(
                    canvas.create_oval(
                        eye_x - 4,
                        eye_y - 4,
                        eye_x + 4,
                        eye_y + 4,
                        fill="#D5B95E",
                        outline=palette["outline"],
                        width=1,
                        tag="snake",
                    )
                )
                self.items.append(
                    canvas.create_oval(
                        eye_x - 1,
                        eye_y - 3,
                        eye_x + 1,
                        eye_y + 3,
                        fill="#171B14",
                        outline="",
                        tag="snake",
                    )
                )
                self.items.append(
                    canvas.create_oval(
                        eye_x - 2,
                        eye_y - 2,
                        eye_x - 1,
                        eye_y - 1,
                        fill="#FFFFFF",
                        outline="",
                        tag="snake",
                    )
                )

            nose_x, nose_y = nose_position
            self.items.append(
                canvas.create_oval(
                    nose_x - 2,
                    nose_y - 2,
                    nose_x + 2,
                    nose_y + 2,
                    fill=palette["outline"],
                    outline="",
                    tag="snake",
                )
            )

def draw_board():
    canvas.delete("board")
    canvas.create_rectangle(
        0,
        0,
        GAME_WIDTH,
        GAME_HEIGHT,
        fill=BACKGROUND_COLOR,
        outline="",
        tag="board",
    )
    rng = random.Random(18)
    patch_colours = ("#29452F", "#2C4A32", "#315035", "#355337", "#263F2D")
    blade_colours = ("#496944", "#547447", "#3A5B39", "#617B4A", "#405F3B")
    for _ in range(34):
        center_x = rng.randint(18, GAME_WIDTH - 18)
        center_y = rng.randint(18, GAME_HEIGHT - 18)
        radius_x = rng.randint(10, 29)
        radius_y = rng.randint(5, 15)
        canvas.create_oval(
            center_x - radius_x,
            center_y - radius_y,
            center_x + radius_x,
            center_y + radius_y,
            fill=rng.choice(patch_colours),
            outline="",
            tag="board",
        )

    for _ in range(520):
        x = rng.randint(4, GAME_WIDTH - 4)
        y = rng.randint(5, GAME_HEIGHT - 5)
        height = rng.randint(3, 9)
        lean = rng.randint(-4, 4)
        canvas.create_line(
            x,
            y,
            x + lean,
            y - height,
            fill=rng.choice(blade_colours),
            width=1 if height < 7 else 2,
            tag="board",
        )

    for _ in range(90):
        x = rng.randint(5, GAME_WIDTH - 5)
        y = rng.randint(5, GAME_HEIGHT - 5)
        canvas.create_oval(
            x,
            y,
            x + 2,
            y + 2,
            fill=rng.choice(("#667A49", "#6A7545", "#405C3A")),
            outline="",
            tag="board",
        )
    canvas.create_rectangle(
        2,
        2,
        GAME_WIDTH - 2,
        GAME_HEIGHT - 2,
        outline="#58734A",
        width=3,
        tag="board",
    )


def change_direction(new_direction):
    global direction
    opposite = {
        "up": "down",
        "down": "up",
        "left": "right",
        "right": "left",
    }
    if new_direction in opposite and opposite[new_direction] != direction:
        direction = new_direction


def check_collision(snake):
    head = snake.coordinates[0]
    return (
        head[0] < 0
        or head[0] >= GAME_WIDTH
        or head[1] < 0
        or head[1] >= GAME_HEIGHT
        or head in snake.coordinates[1:]
    )


def draw_overlay(message, colour):
    canvas.create_text(
        GAME_WIDTH / 2,
        GAME_HEIGHT / 2,
        text=message,
        fill=colour,
        font=("Consolas", 34, "bold"),
        tag="overlay",
    )
    canvas.create_text(
        GAME_WIDTH / 2,
        GAME_HEIGHT / 2 + 52,
        text=f"Best score: {best_score}",
        fill="#EAF6FF",
        font=("Consolas", 16),
        tag="overlay",
    )


def start_game():
    global score, direction, state, skin_shop_window
    if skin_shop_window is not None and skin_shop_window.winfo_exists():
        skin_shop_window.destroy()
        skin_shop_window = None
    score = 0
    direction = "right"
    state = "playing"
    update_shop_button()
    label.config(text="Score: 0")
    status.config(text="Use arrow keys to move")
    canvas.delete("overlay")
    canvas.delete("snake")
    canvas.delete("food")
    draw_board()

    snake = Snake()
    food = Food(snake)
    return snake, food


def end_game():
    global state, best_score
    state = "game_over"
    if score > best_score:
        best_score = score
    save_progress()
    update_shop_button()
    play_sound("game_over")
    draw_overlay("Game Over", "#FFB4B4")
    status.config(text="Press Enter to play again")


def next_turn(snake, food):
    global score, apples

    if state != "playing":
        return

    head_x, head_y = snake.coordinates[0]
    if direction == "up":
        next_head = [head_x, head_y - SPACE_SIZE]
    elif direction == "down":
        next_head = [head_x, head_y + SPACE_SIZE]
    elif direction == "left":
        next_head = [head_x - SPACE_SIZE, head_y]
    else:
        next_head = [head_x + SPACE_SIZE, head_y]

    ate_food = next_head == food.coordinates
    if check_collision(snake) or (
        next_head[0] < 0
        or next_head[0] >= GAME_WIDTH
        or next_head[1] < 0
        or next_head[1] >= GAME_HEIGHT
        or (next_head in snake.coordinates[:-1] and not ate_food)
    ):
        end_game()
        return

    snake.coordinates.insert(0, next_head)
    if ate_food:
        score += 1
        apples += 1
        play_sound("eat")
        label.config(text=f"Score: {score}")
        apples_label.config(text=f"Apples: {apples}")
        save_progress()
        food.change_position(snake)
    else:
        snake.coordinates.pop()

    snake.draw()
    window.after(get_speed(), next_turn, snake, food)


def on_keypress(event):
    global snake, food
    if event.keysym == "Return":
        if state in ("start", "game_over"):
            snake, food = start_game()
            window.after(get_speed(), next_turn, snake, food)
        return

    if state != "playing":
        return

    key = event.keysym.lower()
    if key in ("up", "down", "left", "right"):
        change_direction(key)
